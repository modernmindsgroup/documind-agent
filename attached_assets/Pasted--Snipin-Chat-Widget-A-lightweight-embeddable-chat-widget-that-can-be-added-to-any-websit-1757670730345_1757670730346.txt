    // Add event listeners to the widget elements
    function setupEventListeners(elements) {
      const {
        chatButton,
        chatWindow,
        chatInput,
        sendButton,
        closeButton,
        contactForm,
        submitButton,
        messagesContainer,
        chatInputContainer
      } = elements;
      
      // Toggle chat window when chat button is clicked
      chatButton.addEventListener('click', () => {
        state.isOpen = !state.isOpen;
        
        if (state.isOpen) {
          chatWindow.classList.add('open');
          setTimeout(() => {
            chatInput.focus();
          }, 300);
        } else {
          chatWindow.classList.remove('open');
        }
      });
      
      // Close chat window when close button is clicked
      closeButton.addEventListener('click', () => {
        state.isOpen = false;
        chatWindow.classList.remove('open');
      });
      
      // Adjust textarea height based on content
      chatInput.addEventListener('input', () => {
        chatInput.style.height = 'auto';
        chatInput.style.height = (chatInput.scrollHeight > 120 ? 120 : chatInput.scrollHeight) + 'px';
      });
      
      // Send message when send button is clicked
      sendButton.addEventListener('click', () => {
        sendMessage(elements);
      });
      
      // Send message when Enter key is pressed (but allow Shift+Enter for new line)
      chatInput.addEventListener('keydown', (e) => {
        if (e.key === 'Enter' && !e.shiftKey) {
          e.preventDefault();
          sendMessage(elements);
        }
      });
      
      // Submit contact form
      submitButton.addEventListener('click', () => {
        submitContactForm(elements);
      });
      
      // Submit contact form on Enter key in the last field
      const phoneInput = contactForm.querySelector('#snipin-phone');
      phoneInput.addEventListener('keydown', (e) => {
        if (e.key === 'Enter') {
          e.preventDefault();
          submitContactForm(elements);
        }
      });
    }
    
    // Send message to the API
    function sendMessage(elements) {
      const {
        chatInput,
        messagesContainer,
        contactForm,
        chatInputContainer,
        sendButton
      } = elements;
      
      const messageText = chatInput.value.trim();
      
      if (!messageText) return;
      
      // Add user message to UI
      addMessageToUI('user', messageText, messagesContainer);
      
      // Add user message to state
      state.messages.push({
        type: 'user',
        content: messageText
      });
      
      // Clear input
      chatInput.value = '';
      chatInput.style.height = '40px';
      
      // Show typing indicator
      addTypingIndicator(messagesContainer);
      
      // Disable send button
      sendButton.disabled = true;
      
      // If contact info not submitted yet, show contact form
      if (!state.isContactInfoSubmitted) {
        // Hide chat input
        chatInputContainer.style.display = 'none';
        
        // Show contact form
        contactForm.style.display = 'flex';
        
        // Focus on name input
        setTimeout(() => {
          contactForm.querySelector('#snipin-name').focus();
        }, 100);
        
        // Remove typing indicator after a short delay
        setTimeout(() => {
          removeTypingIndicator(messagesContainer);
          
          // Add bot message prompting for contact info
          addMessageToUI('bot', 'Please provide your contact information to continue.', messagesContainer);
          
          // Add bot message to state
          state.messages.push({
            type: 'bot',
            content: 'Please provide your contact information to continue.'
          });
        }, 1000);
        
        return;
      }
      
      // Send message to API
      sendMessageToAPI(messageText, messagesContainer, sendButton);
    }
    
    // Add message to UI
    function addMessageToUI(type, content, messagesContainer) {
      const messageElement = document.createElement('div');
      messageElement.className = `snipin-message ${type}`;
      messageElement.textContent = content;
      
      messagesContainer.appendChild(messageElement);
      
      // Scroll to bottom
      messagesContainer.scrollTop = messagesContainer.scrollHeight;
    }
    
    // Add typing indicator
    function addTypingIndicator(messagesContainer) {
      const typingIndicator = document.createElement('div');
      typingIndicator.className = 'snipin-typing-indicator';
      typingIndicator.innerHTML = `
        <span></span>
        <span></span>
        <span></span>
      `;
      
      messagesContainer.appendChild(typingIndicator);
      
      // Scroll to bottom
      messagesContainer.scrollTop = messagesContainer.scrollHeight;
    }
    
    // Remove typing indicator
    function removeTypingIndicator(messagesContainer) {
      const typingIndicator = messagesContainer.querySelector('.snipin-typing-indicator');
      
      if (typingIndicator) {
        typingIndicator.remove();
      }
    }
    
    // Submit contact form
    function submitContactForm(elements) {
      const {
        contactForm,
        chatInputContainer,
        messagesContainer,
        sendButton,
        chatInput
      } = elements;
      
      // Get form values
      const name = contactForm.querySelector('#snipin-name').value.trim();
      const email = contactForm.querySelector('#snipin-email').value.trim();
      const phone = contactForm.querySelector('#snipin-phone').value.trim();
      
      // Validate form
      if (!name) {
        alert('Please enter your name.');
        contactForm.querySelector('#snipin-name').focus();
        return;
      }
      
      if (!email) {
        alert('Please enter your email.');
        contactForm.querySelector('#snipin-email').focus();
        return;
      }
      
      if (!isValidEmail(email)) {
        alert('Please enter a valid email address.');
        contactForm.querySelector('#snipin-email').focus();
        return;
      }
      
      // Show typing indicator
      addTypingIndicator(messagesContainer);
      
      // Send contact info to API
      fetch(`${API_BASE_URL}/contact`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          name,
          email,
          phone: phone || null
        })
      })
        .then(response => response.json())
        .then(data => {
          if (data.info === 'success' && data.data && data.data.id) {
            // Set contact ID in state
            state.contactId = data.data.id;
            state.isContactInfoSubmitted = true;
            
            // Hide contact form
            contactForm.style.display = 'none';
            
            // Show chat input
            chatInputContainer.style.display = 'flex';
            
            // Remove typing indicator
            removeTypingIndicator(messagesContainer);
            
            // Add bot message confirming contact info submission
            addMessageToUI('bot', 'Thank you! How can I assist you today?', messagesContainer);
            
            // Add bot message to state
            state.messages.push({
              type: 'bot',
              content: 'Thank you! How can I assist you today?'
            });
            
            // Enable send button
            sendButton.disabled = false;
            
            // Focus on chat input
            setTimeout(() => {
              chatInput.focus();
            }, 100);
            
            // Send the initial message that was waiting
            const initialMessage = state.messages.find(msg => msg.type === 'user');
            if (initialMessage && initialMessage.content) {
              sendMessageToAPI(initialMessage.content, messagesContainer, sendButton);
            }
          } else {
            // Show error message
            removeTypingIndicator(messagesContainer);
            addMessageToUI('bot', 'There was an error submitting your contact information. Please try again.', messagesContainer);
            
            // Add error message to state
            state.messages.push({
              type: 'bot',
              content: 'There was an error submitting your contact information. Please try again.'
            });
            
            // Enable send button
            sendButton.disabled = false;
          }
        })
        .catch(error => {
          console.error('Error submitting contact form:', error);
          
          // Show error message
          removeTypingIndicator(messagesContainer);
          addMessageToUI('bot', 'There was an error submitting your contact information. Please try again.', messagesContainer);
          
          // Add error message to state
          state.messages.push({
            type: 'bot',
            content: 'There was an error submitting your contact information. Please try again.'
          });
          
          // Enable send button
          sendButton.disabled = false;
        });
    }
    
    // Send message to API
    function sendMessageToAPI(messageText, messagesContainer, sendButton) {
      // Prepare request body
      const requestBody = {
        contact_id: state.contactId,
        agent_id: agentId,
        content: messageText
      };
      
      // Add conversation ID if available
      if (state.conversationId) {
        requestBody.conversation_id = state.conversationId;
      }
      
      // Send message to API
      fetch(`${API_BASE_URL}/public/message`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify(requestBody)
      })
        .then(response => response.json())
        .then(data => {
          // Remove typing indicator
          removeTypingIndicator(messagesContainer);
          
          if (data.info === 'success' && data.data) {
            // Save conversation ID if first message
            if (data.data.conversation_id && !state.conversationId) {
              state.conversationId = data.data.conversation_id;
            }
            
            // Add bot response to UI
            addMessageToUI('bot', data.data.content, messagesContainer);
            
            // Add bot response to state
            state.messages.push({
              type: 'bot',
              content: data.data.content
            });
          } else {
            // Show error message
            addMessageToUI('bot', 'Sorry, there was an error processing your message. Please try again.', messagesContainer);
            
            // Add error message to state
            state.messages.push({
              type: 'bot',
              content: 'Sorry, there was an error processing your message. Please try again.'
            });
          }
          
          // Enable send button
          sendButton.disabled = false;
        })
        .catch(error => {
          console.error('Error sending message:', error);
          
          // Remove typing indicator
          removeTypingIndicator(messagesContainer);
          
          // Show error message
          addMessageToUI('bot', 'Sorry, there was an error processing your message. Please try again.', messagesContainer);
          
          // Add error message to state
          state.messages.push({
            type: 'bot',
            content: 'Sorry, there was an error processing your message. Please try again.'
          });
          
          // Enable send button
          sendButton.disabled = false;
        });
    }
    
    // Validate email
    function isValidEmail(email) {
      const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
      return emailRegex.test(email);
    }
    
    // Initialize widget
    function initWidget() {
      const elements = createWidgetDOM();
      setupEventListeners(elements);
    }
    
    // Init when DOM is ready
    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', initWidget);
    } else {
      initWidget();
    }
  })();
